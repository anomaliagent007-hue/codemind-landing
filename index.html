"Bikin fungsi hitung pajak PPN 11% untuk total belanja"

// Codemind mengerti:
def hitung_pajak_ppn(total_belanja: float) -> float:
    """Hitung PPN 11% sesuai tarif berlaku di Indonesia."""
    return total_belanja * 0.11

// Test case ikut dibuat:
def test_hitung_pajak_ppn():
    assert hitung_pajak_ppn(100000) == 11000
    assert hitung_pajak_ppn(0) == 0
